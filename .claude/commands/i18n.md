      height: "1.5rem",

  },

      setWorkoutSatisfaction({
        satisfaction: newSatisfaction,
        currentSlideIndex,
      }),
    );
  };

  return (
    <Box {...other}>
      <BottomSheetSlideLabel>{t("components.BottomSheetReviewSatisfaction.label")}</BottomSheetSlideLabel>
      {/* <BottomSheetSlideSubtitle>{t("components.BottomSheetReviewSatisfaction.subtitle")}</BottomSheetSlideSubtitle> */}
      <StyledRating
        defaultValue={5}
        emptyIcon={<FavoriteBorderIcon sx={{ fontSize: "2rem" }} />}
        getLabelText={(value: number) => `${value} Heart${value !== 1 ? "s" : ""}`}
        icon={<FavoriteIcon fontSize="large" />}
        max={10}
        name="customized-color"
        onChange={handleSatisfactionChange}
        precision={1}

## Pluralisation

**IMPORTANT** : Pour la pluralisation, tu DOIS utiliser `count` comme nom de paramètre. C'est obligatoire pour que i18next sache quelle variante utiliser (`_one` ou `_other`).

Dans le JSON :
```json
{
  "ITEMS_COUNT_one": "{{count}} élément",
  "ITEMS_COUNT_other": "{{count}} éléments"
}
```

Dans le code :
```tsx
// ✅ CORRECT - utilise "count"
t("ITEMS_COUNT", { count: items.length })

// ❌ INCORRECT - ne fonctionne PAS avec un autre nom
t("ITEMS_COUNT", { total: items.length })  // Affichera "ITEMS_COUNT" littéralement
```

Si tu as besoin d'afficher une autre valeur que `count` dans le texte, tu peux passer les deux :
```json
{
  "REMAINING_one": "Plus qu'un élément sur {{total}} !",
  "REMAINING_other": "Plus que {{count}} éléments sur {{total}} !"
}
```

```tsx
t("REMAINING", { count: remaining, total: totalItems })
```

langue anglaise : `./public/locales/en/translation.json`.

Pour les dates et pour connaître la timezone de l'utilisateur, il faut que tu utilises le hook `useTimezone` qui se situe ici :
